
namespace = ecumenical_council

ecumenical_council.0001 = {
	type = character_event
	title = ecumenical_council.0001.t
	desc = ecumenical_council.0001.desc
	theme = realm
	left_portrait = {
		character = root
		animation = personality_zealous
	}

	immediate = {
		play_music_cue = "mx_cue_faith_conversion"
		save_scope_as = council_host
	}

	option = {
		name = ecumenical_council.0001.a
		custom_tooltip = ecumenical_council.0001.a.tt

		set_character_faith = faith:universal_church
		every_held_title = {
			title_tier = county
			set_county_faith = faith:universal_church
		}
		hold_ecumenical_council_scripted_effect = yes
	}
}

ecumenical_council.0002 = {
	type = character_event
	title = ecumenical_council.0002.t
	desc = ecumenical_council.0002.desc
	left_portrait = {
        character = scope:council_host
        animation = personality_zealous
    }
	theme = realm
	immediate = {
        faith = {
            save_scope_as = old_faith
        }
    }

	option = {
		name = ecumenical_council.0002.a
		trigger = {
            OR = {
                is_ai = no
                NOT = {
                    has_trait = zealous
                }
            }
        }
		set_character_faith = faith:universal_church
		every_held_title = {
            title_tier = county
            set_county_faith = faith:universal_church
        }
		ai_chance = {
            base = 80
            modifier = {
                add = 20
                has_trait = cynical
            }
			modifier = {
				add = -10
				highest_held_title_tier >= tier_empire
			}
			modifier = {
				add = 5
				highest_held_title_tier = tier_kingdom
			}
			modifier = {
				add = 10
				highest_held_title_tier = tier_duchy
			}
			modifier = {
				add = 15
				highest_held_title_tier < tier_duchy
			}
        }
	}
	option = {
		name = ecumenical_council.0002.b
		ai_chance = {
			base = 20
			ai_value_modifier = {
                ai_zeal = 1.0
            }
			modifier = {
                add = 20
                has_trait = zealous
            }
		}
	}
}

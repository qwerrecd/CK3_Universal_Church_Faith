# A Grand Ecumenical Council is called

ecumenical_council.0001 = {
	type = character_event
	title = ecumenical_council.0001.t
	desc = ecumenical_council.0001.desc
	left_portrait = {
        character = scope:council_host
        animation = personality_zealous
    }
    right_portrait = {
        character = scope:scoped_pope
        animation = personality_zealous
    }
	theme = realm

	immediate = {
		save_scope_as = council_host
		if = {
			limit = { exists = title:k_papal_state.holder }
			title:k_papal_state.holder = { save_scope_as = scoped_pope }
		}
		if = {
			limit = { exists = title:k_orthodox.holder }
			title:k_orthodox.holder = { save_scope_as = scoped_patriarch }
		}
		if = {
			limit = { exists = title:d_coptic_papacy.holder }
			title:d_coptic_papacy.holder = { save_scope_as = scoped_coptic }
		}
	}
	
	option = {
		name = ecumenical_council.0001.a
		trigger_event = {
			id = ecumenical_council.0100
			days = 14
		}
	}
}

# A new unified faith is proclaimed!

ecumenical_council.0200 = {
	type = character_event
	title = ecumenical_council.0200.t
	desc = ecumenical_council.0200.desc
	theme = realm
	left_portrait = {
		character = root
		animation = personality_zealous
	}

	immediate = {
		play_music_cue = "mx_cue_faith_conversion"
	}

	option = {
		name = ecumenical_council.0200.a
		custom_tooltip = ecumenical_council.0200.a.tt
		hidden_effect = { # no double schism mending
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mended_great_schism
			}
		}
		apply_ecumenical_council_doctrine_effect = yes
		set_character_faith = faith:universal_church
		if = {
			limit = { exists = title:k_papal_state.holder }
			title:k_papal_state.holder = { save_scope_as = scoped_pope }
			destroy_title = title:k_papal_state
			scope:scoped_pope = { # pope converts and the title goes away
				set_character_faith = faith:universal_church
			}
		}
        if = {
			limit = { exists = title:k_orthodox.holder }
			title:k_orthodox.holder = { save_scope_as = scoped_patriarch }
			destroy_title = title:k_orthodox
			scope:scoped_patriarch = { # patriarch converts and the title goes away
				set_character_faith = faith:universal_church
			}
		}
        if = {
			limit = { exists = title:d_coptic_papacy.holder }
			title:d_coptic_papacy.holder = { save_scope_as = scoped_coptic }
			destroy_title = title:d_coptic_papacy
			scope:scoped_coptic = { # coptic pope converts and the title goes away
				set_character_faith = faith:universal_church
			}
		}
		if = {
			limit = { exists = title:e_byzantium.holder }
			title:e_byzantium.holder = {
				set_character_faith = faith:universal_church
			}
			title:e_byzantium = { set_state_faith = faith:universal_church }
		}
		if = {
			limit = { exists = title:e_hre.holder }
			title:e_hre.holder = {
				set_character_faith = faith:universal_church
			}
		}
		every_held_title = {
			title_tier = county
			set_county_faith = faith:universal_church
		}
		hold_ecumenical_council_scripted_effect = yes
	}
}

# Create new head of faith

ecumenical_council.0300 = {
    type = character_event
    title = ecumenical_council.0300.t
    desc = ecumenical_council.0300.desc
	left_portrait = {
        character = root
        animation = personality_zealous
    }
    right_portrait = {
        character = scope:new_apostle
        animation = personality_zealous
    }
    theme = realm

    immediate = {
        play_music_cue = "mx_cue_sacredrite"
        hidden_effect = {
            create_character = {
                employer = root
                template = religious_leader_character
                random_traits = no
                faith = root.faith
                culture = root.culture
                save_scope_as = new_apostle
            }
        }
        scope:new_apostle = {

            add_trait = compassionate
            add_trait = zealous

            add_trait = education_learning_4
            add_trait = theologian

        }
        create_title_and_vassal_change = {
            type = returned
            save_scope_as = change
            add_claim_on_loss = no
        }
        title:k_apostolic_see = {
            change_title_holder = {
                holder = scope:new_apostle
                change = scope:change
            }
        }
        faith:universal_church = {
            set_religious_head_title = title:k_apostolic_see
            change_fervor = {
                value = 15
                desc = fervor_gain_new_apostle
            }
        }
        resolve_title_and_vassal_change = scope:change
    }
    option = {
        name = ecumenical_council.0300.a
    }
}

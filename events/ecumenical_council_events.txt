
namespace = ecumenical_council

ecumenical_council.0001 = {
	type = character_event
	title = ecumenical_council.0001.t
	desc = ecumenical_council.0001.desc
	left_portrait = {
        character = scope:council_host
        animation = personality_zealous
    }
	right_portrait = {
		character = title:k_papal_state.holder
		animation = personality_zealous
	}
	theme = realm

	immediate = {
		save_scope_as = council_host
	}

	option = {
		name = ecumenical_council.0001.a
		ai_chance = {
            base = 50
        }
		faith:universal_church = {
			add_doctrine = tenet_mendicant_preachers
		}
		trigger_event = {
			id = ecumenical_council.0002
			days = 7
		}
	}
	option = {
		name = ecumenical_council.0001.b
		ai_chance = {
			base = 50
		}
		faith:universal_church = {
			add_doctrine = tenet_armed_pilgrimages
		}
		trigger_event = {
			id = ecumenical_council.0002
			days = 7
		}
	}
}

ecumenical_council.0002 = {
	type = character_event
	title = ecumenical_council.0002.t
	desc = ecumenical_council.0002.desc
	theme = realm
	left_portrait = {
		character = root
		animation = personality_zealous
	}

	immediate = {
		play_music_cue = "mx_cue_faith_conversion"
	}

	option = {
		name = ecumenical_council.0002.a
		set_character_faith = faith:universal_church
		every_held_title = {
			title_tier = county
			set_county_faith = faith:universal_church
		}
		hold_ecumenical_council_scripted_effect = yes
	}
}

ecumenical_council.0003 = {
	type = character_event
	title = ecumenical_council.0003.t
	desc = ecumenical_council.0003.desc
	left_portrait = {
        character = scope:council_host
        animation = personality_zealous
    }
	theme = realm

	option = {
		name = ecumenical_council.0003.a
		trigger = {
            OR = {
                is_ai = no
                NOT = {
                    has_trait = zealous
                }
            }
        }
		set_character_faith = faith:universal_church
		every_held_title = {
            title_tier = county
            set_county_faith = faith:universal_church
        }
		ai_chance = {
            base = 85
            modifier = {
                add = 20
                has_trait = cynical
            }
			modifier = {
				add = -10
				highest_held_title_tier >= tier_empire
			}
			modifier = {
				add = 5
				highest_held_title_tier = tier_duchy
			}
			modifier = {
				add = 10
				highest_held_title_tier < tier_duchy
			}
        }
	}
	option = {
		name = ecumenical_council.0003.b
		ai_chance = {
			base = 15
			ai_value_modifier = {
                ai_zeal = 1.0
            }
			modifier = {
                add = 20
                has_trait = zealous
            }
		}
	}
}

establish_apostolic_see_scripted_effect = {

	# Generate the new head of the universal church
	hidden_effect = {
		create_character = {
			employer = root
			template = religious_leader_character
			random_traits = no
			faith = root.faith
			culture = root.culture
			save_scope_as = new_patriarch
		}
	}

	scope:new_patriarch = {
		random_list = {
			1 = { add_trait = education_learning_5 }
			1 = { add_trait = education_learning_4 }
			1 = { add_trait = education_learning_3 }
		}

		random_list = {
			1 = { add_trait = scholar }
			1 = { add_trait = theologian }
		}

		random_list = {
			1 = { add_trait = just }
			1 = { add_trait = compassionate }
			1 = { add_trait = honest }
			1 = { add_trait = forgiving }
		}

		add_trait = zealous
		add_trait = temperate

	}
	title:k_apostolic_see = {
		change_title_holder = {
			holder = scope:new_patriarch
		}
	}
	faith:universal_church = {
		set_religious_head_title = title:k_apostolic_see
	}
}

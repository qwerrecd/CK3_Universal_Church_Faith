hold_ecumenical_council_scripted_effect = {
	every_ruler = { #Notify members of other ecumenical Christian branches that the council has come to an agreement.
		limit = {
			faith.religion = root.faith.religion
			faith != root.faith
			NOT = { government_has_flag = government_is_theocracy }
			this != this.faith.religious_head
			faith = {
				has_doctrine = special_doctrine_ecumenical_christian
			}
		}
		trigger_event = ecumenical_council.0201
	}
    every_ruler = { #Notify members of other Christian branches that they need to get on board or get run over.
		limit = {
			faith.religion = root.faith.religion
			faith != root.faith
			NOT = { government_has_flag = government_is_theocracy }
			this != this.faith.religious_head
			NOT = {
                faith = {
				    has_doctrine = special_doctrine_ecumenical_christian
			    }
            }
		}
		trigger_event = ecumenical_council.0202
	}
	every_player = { #Then notify all infidel players that might be around.
		limit = {
			faith.religion != root.faith.religion
		}
		trigger_event = ecumenical_council.0203
	}

	hidden_effect = {
		faith:universal_church = {
			remove_variable = block_conversion_till_decision_taken
			remove_doctrine = doctrine_invisible_faith
			change_fervor = {
				value = 25
				desc = fervor_gain_mended_schism_ecumenical
			}
		}
		religion:christianity_religion = {
			every_faith = {
				custom = every_ecumenical_christian_faith.tt
				limit = {
					this.religion = root.faith.religion
					has_doctrine = special_doctrine_ecumenical_christian
					this != faith:universal_church
				}
				change_fervor = {
					value = -25
					desc = fervor_loss_mended_schism_ecumenical
				}
			}
		}
		remove_character_flag = ecumenical_council_in_progress
		remove_global_variable = great_ecumenical_council_in_progress
		clear_ecumenical_council_choice_flags_effect = yes
		add_character_flag = held_ecumenical_council
	}

	set_nickname_effect = { NICKNAME = nick_the_ecumenist }
}

apply_ecumenical_council_doctrine_effect = {
	hidden_effect = {
		faith:universal_church = {
			if = {
				limit = { root = { has_character_flag = gec_choice_pluralism_fundamentalist } }
				remove_doctrine = doctrine_pluralism_righteous
				add_doctrine = doctrine_pluralism_fundamentalist
			}
			else_if = {
				limit = { root = { has_character_flag = gec_choice_pluralism_pluralistic } }
				remove_doctrine = doctrine_pluralism_righteous
				add_doctrine = doctrine_pluralism_pluralistic
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_no_ecumenism } }
				remove_doctrine = special_doctrine_ecumenical_christian
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_divorce_disallowed } }
				remove_doctrine = doctrine_divorce_approval
				add_doctrine = doctrine_divorce_disallowed
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_bastardry_none } }
				remove_doctrine = doctrine_bastardry_legitimization
				add_doctrine = doctrine_bastardry_none
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_consanguinity_restricted } }
				remove_doctrine = doctrine_consanguinity_aunt_nephew_and_uncle_niece
				add_doctrine = doctrine_consanguinity_restricted
			}
			else_if = {
				limit = { root = { has_character_flag = gec_choice_consanguinity_cousins } }
				remove_doctrine = doctrine_consanguinity_aunt_nephew_and_uncle_niece
				add_doctrine = doctrine_consanguinity_cousins
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_adultery_men_crime } }
				remove_doctrine = doctrine_adultery_men_shunned
				add_doctrine = doctrine_adultery_men_crime
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_adultery_women_shunned } }
				remove_doctrine = doctrine_adultery_women_crime
				add_doctrine = doctrine_adultery_women_shunned
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_clerical_function_alms } }
				remove_doctrine = doctrine_clerical_function_taxation
				add_doctrine = doctrine_clerical_function_alms_and_pacification
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_clerical_marriage_allowed } }
				remove_doctrine = doctrine_clerical_marriage_disallowed
				add_doctrine = doctrine_clerical_marriage_allowed
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_tenet_mendicant_preachers } }
				add_doctrine = tenet_mendicant_preachers
			}
			else_if = {
				limit = { root = { has_character_flag = gec_choice_tenet_armed_pilgrimages } }
				add_doctrine = tenet_armed_pilgrimages
			}
			if = {
				limit = { root = { has_character_flag = gec_choice_tenet_universal_catechism } }
				add_doctrine = tenet_universal_catechism
			}
			else_if = {
				limit = { root = { has_character_flag = gec_choice_tenet_faith_triumphant } }
				add_doctrine = tenet_faith_triumphant
			}
		}
	}
}

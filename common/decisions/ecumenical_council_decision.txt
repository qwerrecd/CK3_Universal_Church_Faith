hold_ecumenical_council_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds" }
	desc = hold_ecumenical_council_decision_desc
	selection_tooltip = hold_ecumenical_council_decision_tooltip
	decision_group_type = major

    ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 120
		hegemony = 120
	}

	is_shown = {
		is_ruler = yes
		is_playable_character = yes
        is_independent_ruler = yes
        highest_held_title_tier >= tier_kingdom
        NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mended_great_schism
			}
		}
		faith = { has_doctrine = special_doctrine_ecumenical_christian } #They'd never listen to an Adamite calling for this I figure.
	}

	is_valid = {
        # 1. Prestige and Piety
        prestige_level >= 4
		piety_level >= max_piety_level

		# 2. Five Sees controlled by Christendom
        custom_description = {
            text = ecumenical_council_rome_doctrine
            title:c_roma.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }

        custom_description = {
            text = ecumenical_council_constantinople_doctrine
            title:c_byzantion.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }

        custom_description = {
            text = ecumenical_council_jerusalem_doctrine
            title:c_jerusalem.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }

        custom_description = {
            text = ecumenical_council_alexandria_doctrine
            title:c_alexandria.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }

        custom_description = {
            text = ecumenical_council_antioch_doctrine
            title:c_antiocheia.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }


		# 3–7. HRE, ERE, Papacy, Ecumenical Patriarchate, Coptic Papacy
		exists = title:e_hre
		title:e_hre.holder = { 
            opinion = { 
                target = root value >= 75 
            }
        }

		exists = title:e_byzantium
		title:e_byzantium.holder = { 
            opinion = { 
                target = root value >= 75 
            }
        }

        # Catholic Head of Faith Approves
        custom_description = {
            text = ecumenical_papacy_approval
            exists = title:k_papal_state
            title:k_papal_state.holder = {
                OR = {
                    AND = {
                        has_trait = zealous
                        opinion = { target = root value >= 100 }
                    }
                    AND = {
                        NOT = { has_trait = zealous }
                        opinion = { target = root value >= 75 }
                    }
                }
            }
        }

        # Orthodox Head of Faith Approves
        custom_description = {
            text = ecumenical_orthodox_approval
            exists = title:k_orthodox
            title:k_orthodox.holder = {
                OR = {
                    AND = {
                        has_trait = zealous
                        opinion = { target = root value >= 100 }
                    }
                    AND = {
                        NOT = { has_trait = zealous }
                        opinion = { target = root value >= 75 }
                    }
                }
            }
        }

        # Coptic Head of Faith Approves
        custom_description = {
            text = ecumenical_coptic_approval
            exists = title:d_coptic_papacy
            title:d_coptic_papacy.holder = {
                OR = {
                    AND = {
                        has_trait = zealous
                        opinion = { target = root value >= 100 }
                    }
                    AND = {
                        NOT = { has_trait = zealous }
                        opinion = { target = root value >= 75 }
                    }
                }
            }
        }

		# 8. Learning requirement
		OR = {
			learning >= 18
			AND = {
                learning >= 12
                has_trait = theologian
            }
		}

		# 9. Control Jerusalem
		completely_controls = title:k_jerusalem

		# 10. Independent + Ruler tier
		is_independent_ruler = yes
		OR = {
            highest_held_title_tier >= 5
            has_title = title:k_jerusalem
        }
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		top_liege = this
	}

	effect = { 
        add_piety = 1000
        gain_holy_legend_seed_tooltip_effect = yes
        custom_tooltip = hold_ecumenical_council_decision_effect_tooltip
        show_as_tooltip = {
			hold_ecumenical_council_scripted_effect = yes
		}
        trigger_event = ecumenical_council.0001
        hidden_effect = { # no double schism mending
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mended_great_schism
			}
		}
    }

	ai_potential = { 
        always = yes
    }
	ai_will_do = { 
        base = 100 
    }
}

dismantle_papacy_decision = { #For Muslims and Pagans.
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	desc = dismantle_papacy_decision_desc
	selection_tooltip = dismantle_papacy_decision_tooltip
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		is_ruler = yes
		is_playable_character = yes
		faith:catholic = {
			has_doctrine = doctrine_spiritual_head
		}

		faith:catholic = {
			faith_hostility_level = {
				target = root.faith
				value >= faith_hostile_level
			}
            exists = religious_head # dunno why this isn't a check in vanilla but we need it for the edge case
            this.religion != root.faith.religion # this event is clearly designed for non-Christians
		}
		highest_held_title_tier > 3 #Only Kings and Emperors
		capital_province = { #Indians could not possibly care...
			OR = {
				geographical_region = world_europe
				geographical_region = world_middle_east
				geographical_region = world_africa_north
			}
		}
	}

	is_valid = {
		trigger_if = {
			limit = {
				exists = title:k_papal_state.holder
			}
			title:k_papal_state.holder = { is_playable_character = no } #If he fled elsewhere, it cannot be done, the Pope must be landless.
			title:k_papal_state.holder = { is_at_war = no } #Shouldn't be possible during a Crusade
		}
		completely_controls_region = world_europe_south_italy
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		top_liege = this
	}

	effect = {
		custom_tooltip = dismantle_papacy_decision_effect_tooltip
		gain_holy_legend_seed_tooltip_effect = yes
		hidden_effect = {
			legend_seed_great_deed_faith_effect = {
				ANTAGONIST = title:k_papal_state.holder
				OLD_FAITH = title:k_papal_state.holder.faith
			}
		}
		show_as_tooltip = {
			destroy_papacy_scripted_effect = yes
		}
		trigger_event = {
			id = roman_restoration.0130
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = { #Do it always.
		base = 100
	}
}

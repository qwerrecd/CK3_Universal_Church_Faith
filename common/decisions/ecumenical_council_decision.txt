hold_ecumenical_council_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds" }
	desc = hold_ecumenical_council_decision_desc
	selection_tooltip = hold_ecumenical_council_decision_tooltip
	decision_group_type = major

    ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 120
		hegemony = 120
	}

	is_shown = {
		is_ruler = yes
		is_playable_character = yes
        is_independent_ruler = yes
        highest_held_title_tier >= tier_kingdom
        NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mended_great_schism
			}
		}
		faith = { has_doctrine = special_doctrine_ecumenical_christian } #They'd never listen to an Adamite calling for this I figure.
	}

	is_valid = {
        # 1. Prestige and Piety
        prestige_level >= 4
		piety_level >= max_piety_level

		# 2. Five Sees controlled by Christendom
        custom_description = {
            text = ecumenical_council_rome_doctrine
            title:c_roma.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }

        custom_description = {
            text = ecumenical_council_constantinople_doctrine
            title:c_byzantion.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }

        custom_description = {
            text = ecumenical_council_jerusalem_doctrine
            title:c_jerusalem.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }

        custom_description = {
            text = ecumenical_council_alexandria_doctrine
            title:c_alexandria.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }

        custom_description = {
            text = ecumenical_council_antioch_doctrine
            title:c_antiocheia.holder = {
                faith = {
                    has_doctrine = special_doctrine_ecumenical_christian
                }
            }
        }


		# 3–7. HRE, ERE, Papacy, Ecumenical Patriarchate, Coptic Papacy
		exists = title:e_hre
		title:e_hre.holder = { 
            opinion = { 
                target = root value >= 75 
            }
        }

		exists = title:e_byzantium
		title:e_byzantium.holder = { 
            opinion = { 
                target = root value >= 75 
            }
        }

        # Catholic Head of Faith Approves
        custom_description = {
            text = ecumenical_papacy_approval
            exists = title:k_papal_state
            title:k_papal_state.holder = {
                OR = {
                    AND = {
                        has_trait = zealous
                        opinion = { target = root value >= 100 }
                    }
                    AND = {
                        NOT = { has_trait = zealous }
                        opinion = { target = root value >= 75 }
                    }
                }
            }
        }

        # Orthodox Head of Faith Approves
        custom_description = {
            text = ecumenical_orthodox_approval
            exists = title:k_orthodox
            title:k_orthodox.holder = {
                OR = {
                    AND = {
                        has_trait = zealous
                        opinion = { target = root value >= 100 }
                    }
                    AND = {
                        NOT = { has_trait = zealous }
                        opinion = { target = root value >= 75 }
                    }
                }
            }
        }

        # Coptic Head of Faith Approves
        custom_description = {
            text = ecumenical_coptic_approval
            exists = title:d_coptic_papacy
            title:d_coptic_papacy.holder = {
                OR = {
                    AND = {
                        has_trait = zealous
                        opinion = { target = root value >= 100 }
                    }
                    AND = {
                        NOT = { has_trait = zealous }
                        opinion = { target = root value >= 75 }
                    }
                }
            }
        }

		# 8. Learning requirement
		OR = {
			learning >= 18
			AND = {
                learning >= 12
                has_trait = theologian
            }
		}

		# 9. Control Jerusalem
		completely_controls = title:k_jerusalem

		# 10. Independent + Ruler tier
		is_independent_ruler = yes
		OR = {
            highest_held_title_tier >= 5
            has_title = title:k_jerusalem
        }
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		top_liege = this
	}

	effect = { 
        add_piety = 1000
        gain_holy_legend_seed_tooltip_effect = yes
        custom_tooltip = hold_ecumenical_council_decision_effect_tooltip
        show_as_tooltip = {
			hold_ecumenical_council_scripted_effect = yes
		}
        trigger_event = ecumenical_council.0001
        hidden_effect = { # no double schism mending
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_mended_great_schism
			}
		}
    }

	ai_potential = { 
        always = yes
    }
	ai_will_do = { 
        base = 100 
    }
}

create_universal_holy_order_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	decision_group_type = major

	ai_goal = yes

	is_shown = {
		is_landed = yes
    NOR = {
			faith != faith:universal_church # no converting away >:( that was the whole point
			faith = {
				any_faith_holy_order = { # only one holy order allowed
					holy_order_patron = root
					save_temporary_scope_as = order
				}
			}
		}
    has_character_flag = held_ecumenical_council # only the one who held the council can found the order
		exists = faith.religious_head # comes after new religious head is created
	}

	widget = {
		controller = create_holy_order
		barony_valid = {
			trigger_if = {
				limit = { 
					exists = this
			 	}
				barony_is_valid_for_holy_order_lease_trigger = { CHARACTER = scope:ruler }
			}
			trigger_else = {
				custom_description = {
					text = "has_holy_order_barony_selected"
					always = no
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}

	is_valid = {
		trigger_if = {
			custom_description = {
				text = "is_king_or_emperor"
				highest_held_title_tier >= tier_kingdom
			}
		}
		trigger_if = {
			limit = {
				piety_level >= 4
			}
		}
		trigger_if = {
			limit = {
				exists = faith.religious_head
				root != root.faith.religious_head
			}
			faith.religious_head = {
				opinion = {
					target = root
					value >= {
						value = 0
						if = {
							limit = { is_ai = no }
					 		add = high_positive_opinion
						}
					}
				}
			}
		}
		custom_description = {
			text = "has_two_castle_county"
			any_held_county = {
				any_in_de_jure_hierarchy = {
					barony_is_valid_for_holy_order_lease_trigger = { CHARACTER = root }
				}
			}
		}
	}
	cost = {
		gold = {
			value = 50
		}
		treasury = {
			value = 0
			if = {
				limit = {
					has_treasury = yes
				}
				add = 50
			}
		}
		piety = {
			value = 100
		}
	}

	effect = {
		custom_tooltip = create_universal_holy_order_decision_effect_message

		if = { # If you have no valid barony this will throw errors when you open the decision - we add this check to avoid that 
			limit = {
				exists = scope:barony
			}
			hidden_effect = {
				if = {
					limit = { scope:barony.holder != root }
					create_title_and_vassal_change = {
						type = leased_out
						save_scope_as = change
						add_claim_on_loss = no
					}
					scope:barony = {
						change_title_holder_include_vassals = {
							holder = root
							change = scope:change
						}
					}
					resolve_title_and_vassal_change = scope:change
				}
			}

			create_character = {
				template = holy_order_leader_character
				location = scope:barony.title_province
				save_scope_as = leader
			}

			create_holy_order_neutral_effect = {
				LEADER = scope:leader
				CAPITAL = scope:barony
				NEW_HO_SCOPE = new_holy_order
				FOUNDER = root
			}
		}
		create_holy_order_effect = yes
	}
	
	ai_potential = {
		highest_held_title_tier >= tier_kingdom
		is_playable_character = yes
		OR = {
			any_held_title = {
				title_tier = barony
				barony_is_valid_for_holy_order_lease_trigger = { CHARACTER = root }
			}
			any_vassal = {
				primary_title.tier = tier_barony
				any_held_title = {
					barony_is_valid_for_holy_order_lease_trigger = { CHARACTER = root }
				}
			}
		}
	}

	ai_will_do = {
		base = 200
		modifier = {
			add = {
				faith = {
					# Don't create more than 5 holy orders for a faith
					every_faith_holy_order = {
						subtract = 40
					}
				}
			}
		}
		modifier = {
			government_has_flag = government_is_mandala
			add = -150
		}
	}

	confirm_text = create_universal_holy_order_decision_confirm
}
